#!/bin/bash
cd $SLURM_SUBMIT_DIR
echo "running on"
hostname
source /usr/modules/init/bash

module load stringtie

# Define input and output directories
BAM_DIR="/mnt/ceph/bent4061/RNA_seq/7161/for_project/trimmed/mapping_results/sam_files/bam_files/sorted_bam"
# FIXED: full path, so script 02 finds these files regardless of submit folder
OUTPUT_DIR="$BAM_DIR/01_stringtie_files"

# Create output directory if it doesn't exist
mkdir -p $OUTPUT_DIR

# Loop over each BAM file
for BAM_FILE in $BAM_DIR/*.bam; do
    # Extract sample name from the filename
    SAMPLE=$(basename $BAM_FILE .sorted.bam)

    # Run StringTie for transcript assembly
    stringtie -G /mnt/ceph/bent4061/RNA_seq/7161/for_project/trimmed/An_stephensi_genome/genomic.gff -o $OUTPUT_DIR/$SAMPLE.gtf $BAM_FILE
done
echo "finished"

#!/bin/bash
cd $SLURM_SUBMIT_DIR
echo "running on"
hostname
source /usr/modules/init/bash

module load stringtie

# Define input and output directories
BAM_DIR="/mnt/ceph/bent4061/RNA_seq/7161/for_project/trimmed/mapping_results/sam_files/bam_files/sorted_bam"
OUTPUT_DIR="$BAM_DIR/final_stringtie_output"

# Create output directory if it doesn't exist
mkdir -p $OUTPUT_DIR

# Run StringTie using the merged GTF file
for BAM_FILE in $BAM_DIR/*.bam; do
    # Extract sample name from the filename
    # FIXED: strip .sorted.bam (not just .bam) so names match script 01
    SAMPLE=$(basename $BAM_FILE .sorted.bam)

    # FIXED: completed the cut-off command; one folder per sample for Ballgown
    mkdir -p $OUTPUT_DIR/$SAMPLE
    stringtie -B -e -G $BAM_DIR/01_stringtie_files/02_stringtie_files/merged.gtf \
        -o $OUTPUT_DIR/$SAMPLE/$SAMPLE.gtf \
        -A $OUTPUT_DIR/$SAMPLE/${SAMPLE}_gene_abund.tab $BAM_FILE
done

echo "finished"
